cmake_minimum_required(VERSION 3.16)
project(shmem_vector_queues LANGUAGES CXX)
set(SOC_VERSION "Ascend310P3")
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN installation")
set(MEMFABRIC_PREFIX "" CACHE PATH "Matching MemFabric installation")
if(NOT MEMFABRIC_PREFIX)
    message(FATAL_ERROR "Set MEMFABRIC_PREFIX to the matching MemFabric installation")
endif()
include("${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
ascendc_library(vector_queues STATIC kernel.cpp)
ascendc_include_directories(vector_queues PRIVATE "${MEMFABRIC_PREFIX}/include/smem/device")
find_package(MPI REQUIRED COMPONENTS CXX)
add_executable(vector_queue_test main.cpp)
target_compile_features(vector_queue_test PRIVATE cxx_std_17)
target_include_directories(vector_queue_test PRIVATE
    "${CMAKE_BINARY_DIR}/include/vector_queues" "${ASCEND_CANN_PACKAGE_PATH}/include"
    "${MEMFABRIC_PREFIX}/include/smem/host" "${MEMFABRIC_PREFIX}/include/smem/device")
target_link_directories(vector_queue_test PRIVATE "${MEMFABRIC_PREFIX}/lib" "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(vector_queue_test PRIVATE vector_queues MPI::MPI_CXX mf_smem ascendcl runtime dl pthread)
target_link_options(vector_queue_test PRIVATE "-Wl,-rpath-link,${MEMFABRIC_PREFIX}/lib")
